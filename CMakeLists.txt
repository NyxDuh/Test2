cmake_minimum_required(VERSION 3.18)
project(BedrockClient C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/Dobby)
set(IMGUI ${CMAKE_SOURCE_DIR}/third_party/imgui)
add_library(BedrockClient SHARED main.cpp
  ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp
  ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_opengl3.cpp)
target_include_directories(BedrockClient PRIVATE ${IMGUI} ${IMGUI}/backends ${CMAKE_SOURCE_DIR}/third_party/Dobby/include)
target_compile_definitions(BedrockClient PRIVATE IMGUI_IMPL_OPENGL_ES3)
if(TARGET dobby_static)
  target_link_libraries(BedrockClient PRIVATE dobby_static)
else()
  target_link_libraries(BedrockClient PRIVATE dobby)
endif()
target_link_libraries(BedrockClient PRIVATE android log EGL GLESv3 dl)
